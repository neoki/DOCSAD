# ─── FUNCIONES ───────────────────────────────────────────────────────────────
function Clasificar-Archivo {
    param([string]$NombreArchivo)
    
    $nombreLower = $NombreArchivo.ToLower()
    
    foreach ($regla in $Reglas) {
        if ($nombreLower -match $regla.Patron) {
            return $regla.Carpeta
        }
    }
    return $CarpetaDefault
}

# ─── EJECUCIÓN ───────────────────────────────────────────────────────────────
$archivos = Get-ChildItem -Path $SourcePath -File -Recurse | Where-Object {
    $_.Name -match '^\d{3}\.'
}

Write-Host "═══════════════════════════════════════════════════════" -ForegroundColor Cyan
Write-Host "  CLASIFICADOR DE DOCUMENTOS ESCANEADOS" -ForegroundColor Cyan
Write-Host "  Archivos encontrados: $($archivos.Count)" -ForegroundColor Cyan
Write-Host "  Modo: $(if ($Ejecutar) {'EJECUCIÓN REAL'} else {'SIMULACIÓN'})" -ForegroundColor $(if ($Ejecutar) {'Red'} else {'Yellow'})
Write-Host "═══════════════════════════════════════════════════════" -ForegroundColor Cyan

$log = @()
$stats = @{}
$sinCarpeta = @{}
$procesados = 0
$errores = 0

foreach ($archivo in $archivos) {
    $nombre = $archivo.Name
    
    # Extraer código de comunidad (3 dígitos al inicio)
    if ($nombre -match '^(\d{3})\.') {
        $codigo = $Matches[1]
    } else {
        continue
    }
    
    # Buscar carpeta de comunidad
    $carpetaComunidad = $MapeoComunidades[$codigo]
    
    if (-not $carpetaComunidad) {
        # Código no mapeado — registrar para revisión
        if (-not $sinCarpeta[$codigo]) { $sinCarpeta[$codigo] = 0 }
        $sinCarpeta[$codigo]++
        
        $log += [PSCustomObject]@{
            Archivo = $nombre
            Codigo = $codigo
            Comunidad = "SIN MAPEO"
            Tipo = ""
            Destino = ""
            Estado = "SIN_MAPEO"
        }
        continue
    }
    
    # Clasificar tipo de documento
    $tipo = Clasificar-Archivo -NombreArchivo $nombre
    
    # Construir ruta destino
    $rutaDestino = Join-Path $DestPath $carpetaComunidad $tipo
    $archivoDestino = Join-Path $rutaDestino $nombre
    
    # Registrar estadísticas
    $key = "$codigo|$tipo"
    if (-not $stats[$key]) { $stats[$key] = 0 }
    $stats[$key]++
    
    if ($Ejecutar) {
        try {
            # Crear carpeta si no existe
            if (-not (Test-Path $rutaDestino)) {
                New-Item -Path $rutaDestino -ItemType Directory -Force | Out-Null
            }
            
            # Copiar archivo (no mover, por seguridad)
            Copy-Item -Path $archivo.FullName -Destination $archivoDestino -Force
            $procesados++
            
            $log += [PSCustomObject]@{
                Archivo = $nombre
                Codigo = $codigo
                Comunidad = $carpetaComunidad
                Tipo = $tipo
                Destino = $archivoDestino
                Estado = "OK"
            }
        }
        catch {
            $errores++
            $log += [PSCustomObject]@{
                Archivo = $nombre
                Codigo = $codigo
                Comunidad = $carpetaComunidad
                Tipo = $tipo
                Destino = $archivoDestino
                Estado = "ERROR: $_"
            }
        }
    }
    else {
        $procesados++
        $log += [PSCustomObject]@{
            Archivo = $nombre
            Codigo = $codigo
            Comunidad = $carpetaComunidad
            Tipo = $tipo
            Destino = $archivoDestino
            Estado = "SIMULADO"
        }
    }
}

# ─── INFORME ─────────────────────────────────────────────────────────────────
Write-Host ""
Write-Host "═══════════════════════════════════════════════════════" -ForegroundColor Green
Write-Host "  RESULTADO" -ForegroundColor Green
Write-Host "───────────────────────────────────────────────────────"
Write-Host "  Clasificados:    $procesados"
Write-Host "  Errores:         $errores"
Write-Host "  Sin mapeo:       $($log | Where-Object Estado -eq 'SIN_MAPEO' | Measure-Object | Select-Object -ExpandProperty Count)"
Write-Host ""

    Write-Host "  CÓDIGOS SIN MAPEO (revisar manualmente):" -ForegroundColor Yellow
    foreach ($k in $sinCarpeta.Keys | Sort-Object) {
        Write-Host "    Código $k : $($sinCarpeta[$k]) archivos" -ForegroundColor Yellow
    }
}

Write-Host ""
Write-Host "  DISTRIBUCIÓN POR TIPO:" -ForegroundColor Cyan
$tipoStats = @{}
foreach ($k in $stats.Keys) {
    $tipo = $k.Split('|')[1]
    if (-not $tipoStats[$tipo]) { $tipoStats[$tipo] = 0 }
    $tipoStats[$tipo] += $stats[$k]
}
foreach ($t in $tipoStats.Keys | Sort-Object) {
    Write-Host "    $t : $($tipoStats[$t]) archivos"
}

# Guardar log
$log | Export-Csv -Path $LogFile -NoTypeInformation -Encoding UTF8
Write-Host ""
Write-Host "  Log guardado en: $LogFile" -ForegroundColor Gray
Write-Host "═══════════════════════════════════════════════════════" -ForegroundColor Green
